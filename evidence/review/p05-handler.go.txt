func TestReviewDSNTransactionRollbackAndCachedRetry(t *testing.T){
 h,db,id:=reviewDSNSetup(t);_,err:=db.Write().Exec("CREATE TRIGGER review_fail_report BEFORE INSERT ON outgoing_dsn_reports BEGIN SELECT RAISE(ABORT,'review injected marker insert failure'); END");if err!=nil{t.Fatal(err)}
 raw:=dsnReportMIME("retry@example.org","review-env","Final-Recipient: rfc822; good@example.org\r\nAction: delivered\r\nStatus: 2.0.0")
 if c:=reviewDSNOpen(t,h,db,902,raw);c!=200{t.Fatalf("HTTP=%d",c)};rs,_:=db.DSNRecipientsForSend(t.Context(),id);for _,r:=range rs{if r.State!="waiting"{t.Fatal("failed transaction changed recipient")}};var n int;db.Read().QueryRow("SELECT COUNT(*) FROM outgoing_dsn_reports").Scan(&n);if n!=0{t.Fatal("failed transaction left marker")}
 db.Write().Exec("DROP TRIGGER review_fail_report");if c:=reviewDSNReopen(t,h,902);c!=200{t.Fatal(c)};rs,_=db.DSNRecipientsForSend(t.Context(),id);for _,r:=range rs{if r.Address=="good@example.org"&&r.State!="delivered"{t.Fatal(r)}};db.Read().QueryRow("SELECT COUNT(*) FROM outgoing_sends").Scan(&n);if n!=1{t.Fatal("retry added sends")};t.Log("body HTTP=200; failed transaction leaves 0 markers; cached reopen delivered; original outgoing count=1")
}
